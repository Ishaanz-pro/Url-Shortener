USEFUL COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

View Logs:
   docker-compose logs -f web          # API logs
   docker-compose logs -f celery       # Worker logs
   docker-compose logs -f postgres     # Database logs

Stop Services:
   docker-compose down                 # Stop all (keep data)
   docker-compose down -v              # Stop all (remove data)

Restart Services:
   docker-compose restart              # Restart all
   docker-compose restart web          # Restart specific service

Scale Services:
   docker-compose up -d --scale web=3  # Scale API to 3 instances

Execute Commands:
   docker-compose exec web bash        # Bash in API container
   docker-compose exec web pytest      # Run tests

Clean Up:
   docker-compose down -v              # Remove all containers and volumes

TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Problem: Port already in use
Solution: Change port in docker-compose.yml or kill existing process

Problem: Database won't start
Solution: Remove volume and restart
   docker-compose down -v
   docker-compose up -d

Problem: Cannot connect to Redis
Solution: Check Redis service
   docker-compose logs redis

Problem: Celery worker not processing tasks
Solution: Check worker logs
   docker-compose logs celery
